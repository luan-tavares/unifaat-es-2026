+ # Dado, binário e estrutura: tudo vira 0 e 1 (Aulas 03 e 07).

src/10/tipos.py (arquivo novo)
+ # Tipo primitivo x tipo de referência.
+ # Rodar no terminal: python3 tipos.py
+ # Primitivo: a variável guarda o VALOR. Copiar é copiar o valor.
+ a = 10
+ b = a
+ b = b + 1
+ print(a, b) # 10 11 -> mexer em b não afeta a
+ # Referência: a variável guarda o ENDEREÇO do objeto. Copiar é copiar o endereço.
+ x = [1, 2]
+ y = x
+ y.append(3)
+ print(x, y) # [1, 2, 3] [1, 2, 3] -> o mesmo objeto, dois nomes
+ print(x is y) # True
+ # Para ter outro objeto de verdade, é preciso copiar.
+ z = x.copy()
+ z.append(4)
+ print(x, z) # [1, 2, 3] [1, 2, 3, 4]
+ print(x is z) # False

Tudo é binário

O 65 e a letra "A" são os mesmos bits. O que muda é o tipo que o programa usa para ler (Aulas 03 e 07).

Estrutura é organização

bytes([79, 108, 97]) guarda três valores lado a lado. Sabendo onde o primeiro está e quanto cada um ocupa, o endereço de qualquer posição é uma conta, e por isso ler pela posição é rápido.

Primitivo × referência

b = a com inteiro copia o valor; com lista copia só o endereço, e os dois nomes mexem no mesmo objeto (x is y dá True). Copiar de verdade é .copy().

leitura_escrita.py: o antagonismo

A mesma lista de 8 números, solta e ordenada, com a contagem de passos de cada operação.

src/10/leitura_escrita.py (arquivo novo)
+ # Leitura x escrita: a mesma lista de 8 números, organizada de duas formas.
+ # Rodar no terminal: python3 leitura_escrita.py
+ #
+ # O custo é contado em PASSOS. Ganhar na leitura costuma custar na escrita.
+ # --- Lista solta (sem ordem) -----------------------------------------------
+ def escrever_solta(lista, valor):
+ lista.append(valor) # larga no fim
+ return 1 # 1 passo
+ def ler_solta(lista, alvo):
+ passos = 0
+ for item in lista: # sem ordem: só dá para olhar um por um
+ passos = passos + 1
+ if item == alvo:
+ return passos
+ return passos
+ # --- Lista ordenada --------------------------------------------------------
+ def ler_ordenada(lista, alvo):
+ # Busca binária: olha o meio e descarta metade.
+ inicio = 0
+ fim = len(lista) - 1
+ passos = 0
+ while inicio <= fim:
+ meio = (inicio + fim) // 2
+ passos = passos + 1
+ if lista[meio] == alvo:
+ return passos
+ if lista[meio] < alvo:
+ inicio = meio + 1
+ continue
+ fim = meio - 1
+ return passos
+ def escrever_ordenada(lista, valor):
+ # Inserir na posição 0: todos à direita são empurrados uma casa.
+ lista.append(None)
+ passos = 0
+ for i in range(len(lista) - 1, 0, -1):
+ lista[i] = lista[i - 1]
+ passos = passos + 1
+ lista[0] = valor
+ return passos + 1
+ solta = [41, 7, 18, 66, 2, 75, 30, 93]
+ ordenada = [2, 7, 18, 30, 41, 66, 75, 93]
+ print("solta escrever:", escrever_solta(solta, 50)) # 1
+ print("solta ler 93 :", ler_solta(solta, 93)) # 8
+ print("ordenada ler 93 :", ler_ordenada(ordenada, 93)) # 4
+ print("ordenada escrever:", escrever_ordenada(ordenada, 1)) # 9

Custo em passos

Cada função devolve quantos passos gastou. Escrever na solta custa 1, ler custa 8. Na ordenada é o contrário: ler custa 4 (busca binária) e escrever custa 9.

Ganhar de um lado é perder do outro

Manter a ordem dá trabalho na escrita (abrir espaço), e é essa ordem que permite ler descartando metade. Largar solto facilita a escrita e obriga a varrer na leitura.

Busca binária

Olha o meio e descarta metade: inicio, fim e meio andam até achar. Com 1 milhão de itens são uns 20 passos, contra até 1 milhão da varredura.

saída
$ python3 leitura_escrita.py

solta ler 93 : 8
ordenada ler 93 : 4
ordenada escrever: 9

lista.py: o array do Python

Vetor é array. Leitura por posição direta, escrita no fim barata, escrita no início cara.

src/10/lista.py (arquivo novo)
+ # Lista (array / vetor): a primeira estrutura nativa do Python.
+ # Rodar no terminal: python3 lista.py
+ #
+ # Vetor é só outro nome para array. A lista do Python é um array dinâmico:
+ # os itens ficam lado a lado e ela cresce sozinha.
+ notas = [7, 3, 9, 5, 8]
+ # --- Leitura -----------------------------------------------------------------
+ # Por posição: O(1). O endereço é uma conta (início + posição x tamanho).
+ print(notas[3]) # 5
+ print(notas[-1]) # 8: o último
+ print(notas[1:3]) # [3, 9]: fatia, do 1 até antes do 3
+ print(len(notas)) # 5
+ # Por valor: O(n). Sem ordem, só dá para olhar um por um.
+ print(9 in notas) # True
+ print(notas.index(9)) # 2
+ # --- Escrita -----------------------------------------------------------------
+ # No fim: O(1) em média. Cabe no próximo espaço livre.
+ notas.append(6)
+ # No início ou no meio: O(n). Todos os itens à direita andam uma casa.
+ notas.insert(0, 1)
+ print(notas) # [1, 7, 3, 9, 5, 8, 6]
+ notas.remove(3) # remove pelo valor (também varre: O(n))
+ ultima = notas.pop() # remove e devolve o último: O(1)
+ notas.sort() # ordena a própria lista
+ print(notas, ultima) # [1, 5, 7, 8, 9] 6
+ # --- Referência --------------------------------------------------------------
+ # A lista é um tipo de referência: "outra = notas" só dá outro nome.
+ outra = notas
+ outra.append(99)
+ print(notas) # [1, 5, 7, 8, 9, 99]
+ copia = notas.copy() # copiar de verdade cria outro objeto
+ copia.append(100)
+ print(notas == copia, notas is copia) # False False

Leitura

Por posição é O(1) (notas[3]); por valor é O(n) (9 in notas), porque sem ordem só dá para olhar um por um.

Escrita

append cabe no próximo espaço livre (O(1) em média). insert(0, x) empurra todos os itens uma casa: O(n).

Referência

outra = notas não copia: é o mesmo objeto. A lista é um tipo de referência.

tupla.py: imutável

Lê como a lista, não escreve; alterar é criar outra.

src/10/tupla.py (arquivo novo)
+ # Tupla: parecida com a lista, mas imutável.
+ # Rodar no terminal: python3 tupla.py
+ ponto = (10, 20)
+ # --- Leitura: igual à lista --------------------------------------------------
+ print(ponto[1]) # 20: por posição, O(1)
+ x, y = ponto # desempacotamento
+ print(x, y) # 10 20
+ print(ponto.count(10), ponto.index(20)) # 1 1
+ print(len(ponto), 20 in ponto) # 2 True
+ # --- Escrita: não existe -----------------------------------------------------
+ try:
+ ponto[0] = 99
+ except TypeError as erro:
+ print("erro:", erro) # 'tuple' object does not support item assignment
+ # "Alterar" é criar outra tupla, copiando todos os itens: O(n).
+ novo = (99,) + ponto[1:]
+ print(ponto, novo) # (10, 20) (99, 20): a antiga continua intacta
+ # --- O que se ganha com o preço ---------------------------------------------
+ # Imutável: o hash nunca muda, então a tupla pode ser chave de dicionário.
+ mapa = {(10, 20): "casa"}
+ print(mapa[(10, 20)]) # casa
+ # A tupla trava os itens que ela guarda: se um item for uma lista, a lista
+ # continua mutável por dentro.
+ mista = (1, [2, 3])
+ mista[1].append(4)
+ print(mista) # (1, [2, 3, 4])

Sem escrita

ponto[0] = 99 dá TypeError. "Alterar" é montar uma tupla nova copiando tudo: O(n), e a antiga continua intacta.

O que o preço compra

Segurança (ninguém muda o valor por engano) e o hash que nunca muda, então a tupla pode ser chave de dicionário.

Imutável por fora

A tupla trava os itens que guarda. Se um item for uma lista, ela continua mutável por dentro (mista[1].append(4)).

saída
$ python3 tupla.py
20
10 20
1 1
2 True

(10, 20) (99, 20)
casa
(1, [2, 3, 4])

dicionario.py: o famoso HashMap

Chave e valor; o hash diz o slot, sem procurar.

src/10/dicionario.py (arquivo novo)
+ # Dicionário (o famoso HashMap): pares de chave e valor.
+ # Rodar no terminal: python3 dicionario.py
+ #
+ # Por baixo é uma tabela de espalhamento: o hash da chave diz em qual slot o
+ # valor mora, sem procurar.
+ aluno = {"nome": "Ana", "ra": 1234}
+ # --- Leitura por chave: O(1) -------------------------------------------------
+ print(aluno["nome"]) # Ana
+ print(aluno.get("email", "sem e-mail")) # sem e-mail: consulta sem erro
+ print("ra" in aluno) # True: o in olha as chaves
+ # --- Escrita por chave: O(1) -------------------------------------------------
+ aluno["nota"] = 8.5 # inclui
+ aluno["nota"] = 9.0 # altera
+ aluno.update({"curso": "ADS"}) # mistura outro dicionário
+ aluno.pop("ra") # remove a chave
+ # --- Percorrer ---------------------------------------------------------------
+ for chave, valor in aluno.items():
+ print(chave, "->", valor)
+ print(list(aluno.keys())) # ['nome', 'nota', 'curso']
+ print(list(aluno.values())) # ['Ana', 9.0, 'ADS']
+ # --- O preço -----------------------------------------------------------------
+ # Leitura por VALOR: O(n). O hash só ajuda com a chave.
+ print("ADS" in aluno.values()) # True, olhando slot por slot
+ # A chave precisa ser imutável (o hash não pode mudar).
+ try:
+ aluno[[1, 2]] = "lista não pode ser chave"
+ except TypeError as erro:
+ print("erro:", erro) # unhashable type: 'list'

Leitura e escrita por chave

Ambas O(1): o hash da chave calcula em qual slot o valor mora. É o HashMap do Java, o Map do JavaScript, o array associativo do PHP.

O preço

Leitura por valor é O(n) (in aluno.values()): o hash só ajuda com a chave. Além disso a tabela reserva slots vazios (memória) e precisa crescer e reorganizar quando enche.

Chave imutável

aluno[[1, 2]] = ... dá TypeError: unhashable: uma lista pode mudar, então o hash dela não seria estável.

True
nome -> Ana
nota -> 9.0
curso -> ADS
['nome', 'nota', 'curso']
['Ana', 9.0, 'ADS']

pilha.py: o último a entrar é o primeiro a sair

Uma classe que usa uma lista por dentro e só expõe empilhar, desempilhar e topo.

src/10/pilha.py (arquivo novo)
+ # Pilha (stack): o último a entrar é o primeiro a sair (LIFO).
+ # Rodar no terminal: python3 pilha.py
+ #
+ # Não é nativa: é uma classe que usa uma lista por dentro e só expõe os
+ # comandos que definem a pilha. A lista fica privada (encapsulamento, Aula 09).
+ class Pilha:
+ def __init__(self):
+ self.__itens = []
+ def empilhar(self, valor):
+ self.__itens.append(valor) # O(1): só pelo topo
+ def desempilhar(self):
+ if self.vazia():
+ raise IndexError("pilha vazia")
+ return self.__itens.pop() # O(1): o último a entrar sai primeiro
+ def topo(self):
+ return self.__itens[-1] # O(1)
+ def vazia(self):
+ return len(self.__itens) == 0
+ # --- Desfazer (Ctrl+Z) -------------------------------------------------------
+ historico = Pilha()
+ historico.empilhar("digitou A")
+ historico.empilhar("digitou B")
+ historico.empilhar("apagou B")
+ print(historico.desempilhar()) # apagou B
+ print(historico.desempilhar()) # digitou B
+ # --- Leitura x escrita -------------------------------------------------------
+ # Escrever e ler o topo: 1 passo. Ler o fundo: precisa tirar todos de cima.
+ def ler_fundo(pilha):
+ passos = 0
+ while True:
+ passos = passos + 1
+ item = pilha.desempilhar()
+ if pilha.vazia():
+ return item, passos
+ letras = Pilha()
+ for letra in "ABCD":
+ letras.empilhar(letra)
+ print(letras.topo()) # D: 1 passo
+ print(ler_fundo(letras)) # ('A', 4): 4 passos só para ler o fundo
+ # --- A pilha de chamadas: a pilha que o próprio Python usa -------------------
+ def fatorial(n):
+ print("entrou fatorial(", n, ")")
+ if n == 1:
+ print("saiu fatorial( 1 ) = 1")
+ return 1
+ resultado = n * fatorial(n - 1)
+ print("saiu fatorial(", n, ") =", resultado)
+ return resultado
+ fatorial(3)
+ # Entram 3, 2, 1 e saem 1, 2, 3: a última chamada é a primeira a terminar.
+ # Recursão sem condição de parada enche a pilha: RecursionError.

LIFO

empilhar e desempilhar só mexem no topo: O(1). Ctrl+Z e o botão voltar do navegador são pilhas.

Leitura x escrita

Ler o topo é 1 passo; ler o fundo (ler_fundo) exige desempilhar todos os de cima: 4 passos para 4 itens. A pilha ganha simplicidade e perde o acesso livre.

A pilha de chamadas

No fatorial(3), as chamadas entram 3, 2, 1 e saem 1, 2, 3: a última a entrar é a primeira a terminar. Recursão sem parada enche a pilha (RecursionError).

entrou fatorial( 1 )
saiu fatorial( 1 ) = 1
saiu fatorial( 2 ) = 2

fila.py: o primeiro a entrar é o primeiro a sair

Uma classe em cima do deque, que mexe nas duas pontas.

src/10/fila.py (arquivo novo)
+ # Fila (queue): o primeiro a entrar é o primeiro a sair (FIFO).
+ # Rodar no terminal: python3 fila.py
+ #
+ # Não é nativa: é uma classe em cima do deque (collections), feito para
+ # mexer nas duas pontas.
+ from collections import deque
+ class Fila:
+ def __init__(self):
+ self.__itens = deque()
+ def enfileirar(self, valor):
+ self.__itens.append(valor) # O(1): entra pelo fim
+ def desenfileirar(self):
+ if self.vazia():
+ raise IndexError("fila vazia")
+ return self.__itens.popleft() # O(1): sai pela frente
+ def vazia(self):
+ return len(self.__itens) == 0
+ # --- Fila de impressão -------------------------------------------------------
+ impressora = Fila()
+ impressora.enfileirar("relatorio.pdf")
+ impressora.enfileirar("foto.png")
+ impressora.enfileirar("slide.pdf")
+ print(impressora.desenfileirar()) # relatorio.pdf
+ print(impressora.desenfileirar()) # foto.png
+ # --- Leitura x escrita: o antagonismo da lista comum -------------------------
+ # Com lista, tirar o primeiro item (pop(0)) obriga a puxar todos os outros
+ # uma casa para fechar o buraco: O(n). O deque só move o marcador da frente.
+ def passos_pop_zero(lista):
+ passos = 1 # retirar o primeiro
+ lista.pop(0)
+ for _ in lista: # cada item restante é puxado
+ passos = passos + 1
+ return passos
+ print(passos_pop_zero(["A", "B", "C", "D", "E"])) # 5
+ # Ler o meio (fila[2]) também é O(n): o deque caminha desde uma ponta.

FIFO

enfileirar entra pelo fim e desenfileirar sai pela frente, ambos O(1) com o deque.

O antagonismo com a lista

lista.pop(0) puxa todos os itens uma casa para fechar o buraco: 5 passos para 5 itens (O(n)). O popleft() do deque só move o marcador da frente.

O meio

Ler fila[2] é O(n): não há posição direta, o deque caminha desde uma ponta.

arvore.py: hierarquia, um pai por nó

A árvore de arquivos da Aula 08, agora com a análise de leitura e escrita.

src/10/arvore.py (arquivo novo)
+ # Árvore (tree): hierarquia em que cada nó tem um único pai.
+ # Rodar no terminal: python3 arvore.py
+ #
+ # Raiz: o nó sem pai. Folha: o nó sem filhos. É a árvore de arquivos da Aula 08.
+ class No:
+ def __init__(self, nome, pai=None):
+ self.nome = nome
+ self.pai = pai # aponta para cima
+ self.filhos = [] # aponta para baixo
+ if pai is None:
+ return
+ pai.filhos.append(self)
+ def adicionar(self, nome):
+ # Com o pai na mão, um filho novo custa 1 passo: O(1).
+ return No(nome, self)
+ def mover_para(self, novo_pai):
+ # Trocar o pai muda um único ponteiro: O(1), seja qual for o tamanho
+ # do que está dentro da pasta.
+ self.pai.filhos.remove(self)
+ self.pai = novo_pai
+ novo_pai.filhos.append(self)
+ def caminho(self):
+ # Subir pelos pais: um passo por nível (a profundidade).
+ if self.pai is None:
+ return self.nome
+ return self.pai.caminho() + "/" + self.nome
+ def buscar(self, nome):
+ # Sem o caminho, precisa visitar nó por nó: O(n).
+ # Devolve o nó achado e quantos nós foram visitados.
+ visitas = 1
+ if self.nome == nome:
+ return self, visitas
+ for filho in self.filhos:
+ achado, extras = filho.buscar(nome)
+ visitas = visitas + extras
+ if achado is not None:
+ return achado, visitas
+ return None, visitas
+ def imprimir(self, nivel=0):
+ print(" " * nivel + self.nome)
+ for filho in self.filhos:
+ filho.imprimir(nivel + 1)
+ disco = No("C:")
+ users = disco.adicionar("Users")
+ windows = disco.adicionar("Windows")
+ desktop = users.adicionar("Desktop")
+ downloads = users.adicionar("Downloads")
+ sistema = windows.adicionar("Sistema")
+ disco.imprimir()
+ # --- Leitura -----------------------------------------------------------------
+ print(desktop.caminho()) # C:/Users/Desktop (3 níveis)
+ achado, visitas = disco.buscar("Downloads")
+ print(achado.caminho(), visitas) # C:/Users/Downloads 4
+ # --- Escrita -----------------------------------------------------------------
+ desktop.adicionar("Fotos") # 1 passo
+ downloads.mover_para(windows) # 1 ponteiro troca de pai
+ print(downloads.caminho()) # C:/Windows/Downloads

Escrita local barata

adicionar com o pai na mão é 1 passo. mover_para troca um único ponteiro (o pai), não importa quantos arquivos haja dentro da pasta.

Leitura por caminho

grafo.py e grafo_git.py: nós, ligações e o Git

Um dicionário de listas de vizinhos; e o Git como exemplo real, um grafo dirigido acíclico.

src/10/grafo.py (arquivo novo)
+ # Grafo: nós ligados por arestas, sem hierarquia (pode ter ciclo).
+ # Rodar no terminal: python3 grafo.py
+ #
+ # Não é nativo: é uma classe que guarda um dicionário de listas de vizinhos.
+ # A árvore é um grafo sem ciclos e com um único pai por nó.
+ class Grafo:
+ def __init__(self):
+ self.__ligacoes = {}
+ def adicionar_no(self, no):
+ self.__ligacoes.setdefault(no, [])
+ def ligar(self, a, b):
+ # Um item em cada lista: O(1).
+ self.adicionar_no(a)
+ self.adicionar_no(b)
+ self.__ligacoes[a].append(b)
+ self.__ligacoes[b].append(a)
+ def vizinhos(self, no):
+ # A lista já está pronta no dicionário: O(1).
+ return self.__ligacoes[no]
+ def ligados(self, a, b):
+ # Varre a lista de vizinhos de a: O(grau).
+ return b in self.__ligacoes[a]
+ def remover(self, a, b):
+ # Acha b na lista de a e a na lista de b: O(grau).
+ self.__ligacoes[a].remove(b)
+ self.__ligacoes[b].remove(a)
+ amigos = Grafo()
+ amigos.ligar("Ana", "Bia")
+ amigos.ligar("Ana", "Caio")
+ amigos.ligar("Bia", "Caio") # fecha um ciclo: Ana, Bia, Caio
+ amigos.ligar("Caio", "Duda")
+ # --- Leitura -----------------------------------------------------------------
+ print(amigos.vizinhos("Caio")) # ['Ana', 'Bia', 'Duda']
+ print(amigos.ligados("Ana", "Duda")) # False
+ # --- Escrita -----------------------------------------------------------------
+ amigos.ligar("Ana", "Duda")
+ print(amigos.ligados("Ana", "Duda")) # True
+ amigos.remover("Ana", "Caio")
+ print(amigos.vizinhos("Ana")) # ['Bia', 'Duda']
src/10/grafo_git.py (arquivo novo)
+ # O Git é um grafo: cada commit é um nó e aponta para o(s) seu(s) pai(s).
+ # Rodar no terminal: python3 grafo_git.py
+ #
+ # As setas só apontam para trás e nunca fecham um ciclo: é um grafo dirigido
+ # acíclico (DAG). Branch é só um ponteiro para um commit. Merge é um commit
+ # com DOIS pais, e é isso que impede o histórico de ser uma lista.
+ pais = {} # commit -> lista dos seus pais (as arestas)
+ branches = {} # branch -> commit para onde ela aponta
+ def commitar(nome, lista_de_pais):
+ # Escrever é barato: um nó novo e uma aresta para cada pai (O(1)).
+ pais[nome] = list(lista_de_pais)
+ commitar("A", [])
+ commitar("B", ["A"])
+ branches["main"] = "B"
+ commitar("C", [branches["main"]]) # git commit na main
+ branches["main"] = "C"
+ branches["feature"] = "B" # git switch -c feature (só um ponteiro)
+ commitar("D", [branches["feature"]]) # git commit na feature
+ branches["feature"] = "D"
+ commitar("E", [branches["main"], branches["feature"]]) # git merge feature
+ branches["main"] = "E"
+ def historico(commit):
+ # Ler o histórico é seguir os pais, um commit por passo (git log).
+ visitados = []
+ fila = [commit]
+ while len(fila) > 0:
+ atual = fila.pop(0)
+ if atual in visitados:
+ continue
+ visitados.append(atual)
+ fila.extend(pais[atual])
+ return visitados
+ print(pais)
+ # {'A': [], 'B': ['A'], 'C': ['B'], 'D': ['B'], 'E': ['C', 'D']}
+ print(branches) # {'main': 'E', 'feature': 'D'}
+ print(historico(branches["main"])) # ['E', 'C', 'D', 'B', 'A']
+ merges = [c for c in pais if len(pais[c]) > 1]
+ print(merges) # ['E']: o commit com dois pais

Dicionário de listas

A chave é o nó, o valor é a lista de vizinhos. ligar e vizinhos são O(1); ligados e remover varrem a lista: O(grau).

A árvore é um grafo

Com ciclo (Ana, Bia, Caio) já não é árvore. Árvore é o grafo sem ciclos e com um único pai por nó.

O Git é um grafo

Commit é nó, "pai" é aresta, branch é só um ponteiro, e o merge é um commit com dois pais (E aponta para C e D). Como as setas só vão para trás, é um DAG. git commit é O(1); git log segue os pais: O(n).
