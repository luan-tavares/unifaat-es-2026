+ # Regra 1: um único nível de indentação por método.
+ # Rodar no terminal: python3 01_indentacao.py
+ turma = {
+ "Ana": [8, 9, 7],
+ "Bruno": [4, 5, 6],
+ "Carla": [],
+ "Diego": [10, 9, 10],

+ def aprovados_antes(turma):

O hadouken

certificado_antes empilha quatro if e o que importa ("certificado emitido") fica no fundo da seta. Para chegar lá, o leitor guardou quatro condições na cabeça.

Fail first

certificado testa o que está ERRADO primeiro e sai cedo (cláusula de guarda). Tudo no nível 1, e cada saída diz o motivo: a saída mostra negado: frequência abaixo de 75%.

Extrair o aninhado

aprovados_antes mistura percorrer, somar, calcular a média e decidir em três níveis. Com media, foi_aprovado e aprovados, cada função faz uma coisa, e o nome vira a explicação.

02_sem_else.py: três saídas no lugar do else

Regra 2 (Don't use the ELSE keyword). O else costuma ser sintoma de uma unidade fazendo coisa demais.

src/11/02_sem_else.py (arquivo novo)
+ # Regra 2: não usar else. Três saídas no lugar dele.
+ # Rodar no terminal: python3 02_sem_else.py
+ # 1) Retorno antecipado (fail first / cláusula de guarda): cada caso resolve e sai.
+ def conceito(nota):
+ if nota >= 9:
+ return "A"
+ if nota >= 7:
+ return "B"
+ if nota >= 5:
+ return "C"
+ return "D"
+ # 2) Valor padrão antes do if: o if só sobrescreve quando precisa.
+ def situacao(media):
+ resultado = "reprovado"
+ if media >= 6:
+ resultado = "aprovado"
+ return resultado
+ # 3) Polimorfismo (Aula 09): cada classe sabe o seu desconto, ninguém pergunta o tipo.
+ class Aluno:
+ def desconto(self):
+ return 0.5
+ class Professor:
+ def desconto(self):
+ return 0.3
+ class Visitante:
+ def desconto(self):
+ return 0.0
+ def preco_final(preco, cliente):
+ return preco * (1 - cliente.desconto())
+ print(conceito(9.5), conceito(7), conceito(3)) # A B D
+ print(situacao(8), situacao(4)) # aprovado reprovado
+ print(preco_final(100, Aluno()), preco_final(100, Visitante())) # 50.0 100.0
saída
$ python3 02_sem_else.py
A B D
aprovado reprovado
50.0 100.0

Retorno antecipado

conceito resolve cada caso e sai (fail first). O leitor não precisa lembrar das condições de cima para entender o return "D" do fim.

Valor padrão antes do if

situacao começa com "reprovado" e o if só sobrescreve quando precisa. Uma variável, um if, nenhum else.

Polimorfismo (Aula 09)

Cada classe (Aluno, Professor, Visitante) sabe o seu desconto(). preco_final não pergunta o tipo: tipo novo é classe nova, e a função antiga não muda.

04_colecao_primeira_classe.py: quem tem uma lista não tem mais nada

Regra 4 (First-class collections). A lista ganha um dono, que oferece só as operações que fazem sentido.

src/11/04_colecao_primeira_classe.py (arquivo novo)
+ # Regra 4: coleção de primeira classe. Quem tem uma lista não tem mais nada.
+ # Rodar no terminal: python3 04_colecao_primeira_classe.py
+ class Aluno:
+ def __init__(self, nome, nota):
+ self._nome = nome
+ self._nota = nota
+ def aprovado(self):
+ return self._nota >= 6
+ def descrever(self):
+ return self._nome + ": " + str(self._nota)
+ class Turma:
+ def __init__(self, alunos=()):
+ self._alunos = list(alunos)
+ def matricular(self, aluno):
+ self._alunos.append(aluno)
+ def aprovados(self):
+ return Turma([aluno for aluno in self._alunos if aluno.aprovado()])
+ def quantidade(self):
+ return len(self._alunos)
+ def descrever(self):
+ return "\n".join(aluno.descrever() for aluno in self._alunos)
+ turma = Turma()
+ turma.matricular(Aluno("Ana", 8.5))
+ turma.matricular(Aluno("Bruno", 4))
+ turma.matricular(Aluno("Diego", 10))
+ print(turma.aprovados().descrever())
+ # Ana: 8.5
+ # Diego: 10
+ print(turma.aprovados().quantidade(), "de", turma.quantidade()) # 2 de 3
saída
$ python3 04_colecao_primeira_classe.py

A lista tem dono

Turma guarda _alunos e nada mais. Ninguém de fora recebe a lista crua (o vazamento da lista da Aula 09).

Devolve outra Turma

aprovados() não devolve uma lista: devolve uma nova Turma, que já sabe quantidade() e descrever(). A estrutura de dados da Aula 10 vira classe do domínio.

05_um_ponto_por_linha.py: Lei de Demeter, não fale com estranhos

Regra 5 (One dot per line, Law of Demeter). A mesma árvore de arquivos da Aula 08, agora perguntando ao amigo certo.

src/11/05_um_ponto_por_linha.py (arquivo novo)
+ # Regra 5: um ponto por linha (Lei de Demeter: fale só com os seus amigos).
+ # Rodar no terminal: python3 05_um_ponto_por_linha.py
+ # ANTES: quem chama conhece a estrutura inteira da árvore da Aula 08.
+ class ItemAberto:
+ def __init__(self, nome, pai=None):
+ self.nome = nome
+ self.pai = pai
+ disco = ItemAberto("C:")
+ usuarios = ItemAberto("Users", disco)
+ desktop = ItemAberto("Desktop", usuarios)
+ trabalhos = ItemAberto("Trabalhos", desktop)
+ tf = ItemAberto("tf-08.py", trabalhos)
+ print(tf.pai.pai.pai.pai.nome) # C: (quatro pontos: se a árvore mudar, quebra aqui)
+ # DEPOIS: o item só fala com o pai direto e o pai responde pelo resto.
+ class Item:
+ def __init__(self, nome, pai=None):
+ self._nome = nome
+ self._pai = pai
+ def disco(self):
+ if self._pai is None:
+ return self._nome
+ return self._pai.disco()
+ raiz = Item("C:")
+ pasta = Item("Users", raiz)
+ arquivo = Item("tf-08.py", pasta)
+ print(arquivo.disco()) # C: (um ponto)
+ # Exceção: encadear métodos que devolvem o mesmo tipo (fluente) não é navegar.
+ print(" Ola ".strip().lower()) # ola
saída
$ python3 05_um_ponto_por_linha.py
C:
C:
ola

Quatro pontos, quatro estranhos

tf.pai.pai.pai.pai.nome atravessa quatro objetos: quem chama conhece o desenho inteiro da árvore, e uma árvore com um nível a mais quebra todo lugar que copiou a cadeia.

Delegar ao pai

arquivo.disco() fala só com o pai direto; o pai repete o gesto até a raiz, que responde com o próprio nome. Um ponto, e a resposta continua certa se a árvore crescer.

A exceção

Encadear métodos que devolvem o mesmo tipo (" Ola ".strip().lower()) não é navegar entre objetos, é uma sequência de transformações.

06_diga_nao_pergunte.py: sem getters e setters

Regra 9 do texto original (No getters/setters/properties), conhecida como Tell, Don't Ask.

src/11/06_diga_nao_pergunte.py (arquivo novo)
+ # Regra 9: sem getters/setters/properties. Diga ao objeto o que fazer, não pergunte.
+ # Rodar no terminal: python3 06_diga_nao_pergunte.py
+ # ANTES: o objeto entrega o saldo e quem chama decide a regra.
+ class ContaAberta:
+ def __init__(self, saldo):
+ self._saldo = saldo
+ def get_saldo(self):
+ return self._saldo
+ def set_saldo(self, valor):
+ self._saldo = valor
+ conta = ContaAberta(100)
+ if conta.get_saldo() >= 30: # a regra mora fora da conta
+ conta.set_saldo(conta.get_saldo() - 30) # e cada chamador pode esquecê-la
+ conta.set_saldo(-999) # nada impede isto
+ # DEPOIS: a regra mora na conta, e só existe o comportamento.
+ class Conta:
+ def __init__(self, saldo):
+ self._saldo = saldo
+ def sacar(self, valor):
+ if valor > self._saldo:
+ raise ValueError("saldo insuficiente")
+ self._saldo -= valor
+ def descrever(self):
+ return "saldo: " + str(self._saldo)
+ conta = Conta(100)
+ conta.sacar(30)
+ print(conta.descrever()) # saldo: 70
+ try:
+ conta.sacar(500)
+ except ValueError as erro:
+ print("Recusou:", erro) # Recusou: saldo insuficiente
saída
$ python3 06_diga_nao_pergunte.py
saldo: 70
Recusou: saldo insuficiente

Perguntar e decidir fora

Com get_saldo e set_saldo, a regra ("só saca se tiver saldo") mora em quem chama, e conta.set_saldo(-999) nada impede.

Dizer e a conta decide

conta.sacar(30) entrega a ordem e a Conta aplica a própria regra (saldo insuficiente na recusa). Um @property de leitura seria só um getter com outra roupa.

Aula 09 x Aula 11

Na Aula 09 o getter/setter era "a porta controlada". Aqui vai um passo além: o objeto não entrega o estado cru, ele oferece comportamento.

08_divida_tecnica.py: o atalho consciente

Entregar rápido é um empréstimo: pega-se velocidade agora e paga-se depois, com juros. Não tem problema, desde que a gente saiba.

src/11/08_divida_tecnica.py (arquivo novo)
+ # Dívida técnica: entregar rápido é um empréstimo. Pode, desde que a gente saiba.
+ # Rodar no terminal: python3 08_divida_tecnica.py
+ # DÍVIDA TÉCNICA (consciente), sexta 17h:
+ # preco_professor é cópia de preco_aluno.
+ # Pagar: unificar em um dicionário de multiplicadores (ticket #42, sprint 3).
+ def preco_aluno(preco):
+ return preco * 0.5
+ def preco_professor(preco):
+ return preco * 0.7
+ print(preco_aluno(100), preco_professor(100)) # 50.0 70.0
+ # Pagamento (sprint 3): uma função, um dicionário, nada duplicado.
+ MULTIPLICADOR = {"aluno": 0.5, "professor": 0.7}
+ def preco_final(preco, tipo):
+ return preco * MULTIPLICADOR[tipo]
+ print(preco_final(100, "aluno"), preco_final(100, "professor")) # 50.0 70.0
saída
$ python3 08_divida_tecnica.py
50.0 70.0
50.0 70.0

Dívida registrada

O comentário diz o quê, por quê e quando pagar (ticket, sprint 3). A mesma duplicação, sem o comentário, seria um descuido: os juros cresceriam em silêncio.

O pagamento

MULTIPLICADOR e preco_final removem a duplicação. A saída é igual antes e depois, que é o que torna seguro pagar a dívida.

09_modularizacao.py: o guarda-chuva das regras

O que une todas as regras: separar as unidades de execução. É a resposta para "por que não posso usar else?".

src/11/09_modularizacao.py (arquivo novo)
+ # Modularização: o guarda-chuva das sete regras. Separar as unidades de execução.
+ # Rodar no terminal: python3 09_modularizacao.py
+ turma = {
+ "Ana": [8, 9, 7],
+ "Bruno": [4, 5, 6],
+ "Carla": [],
+ "Diego": [10, 9, 10],
+ }
+ # ANTES: uma unidade só faz tudo (percorre, soma, calcula a média, decide e imprime).
+ def relatorio_antes(turma):
+ for nome in turma:
+ if len(turma[nome]) > 0:
+ soma = 0
+ for nota in turma[nome]:
+ soma += nota
+ media = soma / len(turma[nome])
+ if media >= 6:
+ print(nome, "aprovado")
+ # DEPOIS: cada função é uma unidade com uma responsabilidade, e a pilha de chamadas
+ # vira uma história: relatorio -> foi_aprovado -> media.
+ def media(notas):
+ return sum(notas) / len(notas)
+ def foi_aprovado(notas):
+ return len(notas) > 0 and media(notas) >= 6
+ def relatorio(turma):
+ for nome in turma:
+ if foi_aprovado(turma[nome]):
+ print(nome, "aprovado")
+ relatorio_antes(turma) # Ana aprovado / Diego aprovado
+ relatorio(turma) # Ana aprovado / Diego aprovado
saída
$ python3 09_modularizacao.py
Ana aprovado
Diego aprovado
Ana aprovado
Diego aprovado

Uma unidade, cinco tarefas

relatorio_antes percorre, soma, calcula a média, decide e imprime: na pilha de execução, é um frame só carregando cinco responsabilidades.

A pilha vira uma história

Com media, foi_aprovado e relatorio, a pilha mostra relatorio chamando foi_aprovado, que chama media. Se algo quebrar, o stack trace aponta o culpado, com nome e uma responsabilidade só.

Obsessão por tipos primitivos

CPF, email e CEP são str, o dinheiro é float e o status é uma letra ("A", "P", "C"): nenhum tem regra.

Sem normalização

Os dados da cliente Ana se repetem em cada pedido dela, e o preço do Fone aparece em dois itens. Cada informação deveria morar em um lugar só.

Hadouken e else

valida empilha cinco if com else; desc e st_txt são escadas de if, elif e else sobre uma letra.

Lista solta e getters

ped é uma lista global, os itens são listas cruas e Ped tem get_st/set_st com dez atributos públicos abreviados.

A rede de segurança

O TF pede para guardar esta saída (saida-original.txt) e rodar um diff a cada passo: refatorar é mudar a estrutura sem mudar o comportamento.
