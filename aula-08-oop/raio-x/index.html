funcoes.py: entrada, saída e void

Um tópico só sobre funções, antes de colocar funções dentro de classes. O código da aula fica em src/08/, separado do src/07/ (ainda não há projeto unificado).

es-2026/ ├── aula-08-oop/ // briefing, slide e este raio-x └── src/ ├── 07/ // estruturas, lacos, binario, merge_sort (aula 07) └── 08/ ← NOVO: uma pasta por aula ├── funcoes.py // entrada, saída e void ├── classes.py // molde, objeto, propriedades, métodos e self ├── construtor.py // __init__ e objeto dentro de objeto └── arvore_arquivos.py // a classe Item: pasta ou arquivo
src/08/funcoes.py (arquivo novo)
+ # Funções: entrada (parâmetros), processamento e saída (return).
+ # Rodar no terminal: python3 funcoes.py
+ #
+ # entrada (parâmetros) -> [ corpo da função ] -> saída (return)
+ # --- Parâmetro e argumento ---------------------------------------------------
+ # PARÂMETRO é o nome que a função declara para receber um valor.
+ # ARGUMENTO é o valor de verdade, passado na hora de chamar.
+ def calcular_media(primeira_nota, segunda_nota):
+ soma = primeira_nota + segunda_nota
+ media = soma / 2
+ return media # a SAÍDA: devolve o valor para quem chamou
+ # 8 e 6 são os argumentos: entram em primeira_nota e segunda_nota.
+ resultado = calcular_media(8, 6)
+ print(resultado) # 7.0
+ # Os argumentos também podem ser variáveis, e a ordem importa.
+ nota_a = 9
+ nota_b = 5
+ print(calcular_media(nota_a, nota_b)) # 7.0
+ # --- return x print ----------------------------------------------------------
+ # print só MOSTRA na tela; quem chamou não recebe nada.
+ # return DEVOLVE o valor: dá para guardar numa variável e usar depois.
+ def media_que_so_mostra(primeira_nota, segunda_nota):
+ print((primeira_nota + segunda_nota) / 2)
+ valor = media_que_so_mostra(8, 6) # mostra 7.0 na tela
+ print(valor) # None: a função não devolveu nada
+ # --- Sem return: o "void" (procedure) ----------------------------------------
+ # Em C, uma função que não devolve nada é void. Em Python, uma função sem
+ # return devolve None, o valor que representa "nada".
+ # Uma função que só FAZ alguma coisa (mostrar, gravar, enviar) e não devolve
+ # resultado é chamada de procedimento (procedure).
+ def mostrar_boletim(nome, media):
+ print("Aluno:", nome, "| Média:", media)
+ mostrar_boletim("Ana", 7.0)
+ # --- Sem parâmetros ----------------------------------------------------------
+ # A entrada é opcional: a função pode não receber nada.
+ def mostrar_cabecalho():
+ print("=== UNIFAAT 2026.2 ===")
+ mostrar_cabecalho()
+ # --- Valor padrão ------------------------------------------------------------
+ # Se o argumento não for passado, o parâmetro assume o valor padrão.
+ def aprovado(media, media_minima=7):
+ return media >= media_minima
+ print(aprovado(8)) # True (usou a média mínima 7)
+ print(aprovado(6)) # False
+ print(aprovado(6, 5)) # True (a média mínima passou a ser 5)
+ # --- Mais de um retorno ------------------------------------------------------
+ # Python devolve uma tupla (Aula 07), e dá para desempacotar na chamada.
+ def menor_e_maior(valores):
+ menor = valores[0]
+ maior = valores[0]
+ for valor in valores:
+ if valor < menor:
+ menor = valor
+ if valor > maior:
+ maior = valor
+ return menor, maior
+ minimo, maximo = menor_e_maior([7, 3, 9, 1, 5])
+ print(minimo, maximo) # 1 9
+ # --- Retorno antecipado ------------------------------------------------------
+ # O return encerra a função na hora: cada condição resolve o seu caso e sai.
+ def situacao(media):
+ if media >= 7:
+ return "Aprovado"
+ if media >= 5:
+ return "Recuperação"
+ return "Reprovado"
+ print(situacao(calcular_media(8, 6))) # Aprovado

Parâmetro × argumento

Parâmetro é o nome na declaração (primeira_nota); argumento é o valor passado na chamada (calcular_media(8, 6)). A ordem importa.

return × print

return devolve o valor e encerra a função; print só mostra na tela. Por isso print(valor) imprime None: a função não devolveu nada.

void é procedure

Em C, sem retorno é void. Em Python não existe a palavra: função sem return devolve None. Função que só faz algo (mostrar, gravar) é um procedimento.

Etapa 2

classes.py: molde, objeto e self

A classe Aluno, dois objetos criados do mesmo molde e o self separando os dados de cada um.

src/08/classes.py (arquivo novo)
+ # Classe e objeto: propriedades, métodos e o self.
+ # Rodar no terminal: python3 classes.py
+ #
+ # A CLASSE é o molde: descreve quais propriedades e métodos o dado terá.
+ # O OBJETO é o dado criado a partir do molde.
+ #
+ # Em C, a struct só agrupa propriedades. Na classe, o mesmo bloco agrupa
+ # também as funções que trabalham com essas propriedades: os MÉTODOS.
+ class Aluno:
+ # Método: uma função declarada dentro da classe.
+ # O primeiro parâmetro é sempre o self: o próprio objeto que está
+ # chamando o método. É por ele que o método sabe de QUEM é a nota.
+ def adicionar_nota(self, nota):
+ self.notas.append(nota)
+ def calcular_media(self):
+ soma = 0
+ for nota in self.notas:
+ soma = soma + nota
+ return soma / len(self.notas)
+ def situacao(self):
+ media = self.calcular_media()
+ if media >= 7:
+ return "Aprovado"
+ if media >= 5:
+ return "Recuperação"
+ return "Reprovado"
+ # --- Criando objetos a partir do molde ---------------------------------------
+ # Aluno() cria um objeto novo, vazio. Cada chamada cria um objeto diferente.
+ ana = Aluno()
+ bruno = Aluno()
+ # Propriedade: um dado que pertence ao objeto. Pode ser de qualquer tipo.
+ # Aqui criamos as propriedades uma a uma, depois que o objeto já existe.
+ ana.nome = "Ana" # str
+ ana.ra = 1234 # int
+ ana.notas = [] # list
+ bruno.nome = "Bruno"
+ bruno.ra = 5678
+ bruno.notas = []
+ # --- Chamando métodos --------------------------------------------------------
+ # Em ana.adicionar_nota(8), o Python passa a ana como self automaticamente.
+ # Dentro do método, self.notas é a lista da ana, e não a do bruno.
+ ana.adicionar_nota(8)
+ ana.adicionar_nota(9)
+ bruno.adicionar_nota(4)
+ bruno.adicionar_nota(6)
+ print(ana.nome, ana.calcular_media(), ana.situacao()) # Ana 8.5 Aprovado
+ print(bruno.nome, bruno.calcular_media(), bruno.situacao()) # Bruno 5.0 Recuperação
+ # --- Mesmo molde, objetos independentes --------------------------------------
+ # Mudar um objeto não mexe no outro: cada um guarda os seus próprios dados.
+ print(ana.notas) # [8, 9]
+ print(bruno.notas) # [4, 6]
+ print(ana is bruno) # False: são dois objetos diferentes
+ # Problema deste jeito: se esquecermos de criar a propriedade notas,
+ # o adicionar_nota quebra com AttributeError. O molde deveria garantir que
+ # todo objeto já nasça com as propriedades certas. Isso é o construtor.

Classe é o molde, objeto é o dado

Aluno descreve como o aluno vai ser. ana e bruno são dois objetos diferentes, criados a partir do mesmo molde (ana is bruno dá False).

Propriedade e método

Propriedade é o que o objeto tem (nome, ra, notas: texto, número e lista). Método é o que ele faz: uma função declarada dentro da classe.

O self

Em ana.adicionar_nota(8), o Python entrega a ana ao parâmetro self. Dentro do método, self.notas é a lista da Ana, não a do Bruno: o mesmo código serve a todos os objetos sem perder o dono de cada dado.

Etapa 3

construtor.py: __init__ e objeto dentro de objeto

O construtor faz o objeto nascer completo, e a propriedade pode guardar outro objeto.

src/08/construtor.py (arquivo novo)
+ # Construtor: __init__ e objeto dentro de outro objeto.
+ # Rodar no terminal: python3 construtor.py
+ # --- __init__: o construtor --------------------------------------------------
+ # O __init__ é um método especial, chamado automaticamente no instante em que
+ # o objeto é criado. Os argumentos passados em Aluno(...) chegam nele.
+ # Serve para o objeto já nascer com as suas propriedades preenchidas.
+ class Aluno:
+ def __init__(self, nome, ra):
+ self.nome = nome
+ self.ra = ra
+ self.notas = [] # toda lista começa vazia, uma para cada objeto
+ def adicionar_nota(self, nota):
+ self.notas.append(nota)
+ def calcular_media(self):
+ soma = 0
+ for nota in self.notas:
+ soma = soma + nota
+ return soma / len(self.notas)
+ # Antes: Aluno() e depois ana.nome = "Ana", ana.ra = 1234, ana.notas = [].
+ # Agora: uma linha só, e nenhuma propriedade fica faltando.
+ ana = Aluno("Ana", 1234)
+ ana.adicionar_nota(8)
+ ana.adicionar_nota(9)
+ print(ana.nome, ana.calcular_media()) # Ana 8.5
+ # --- Um objeto dentro de outro objeto ----------------------------------------
+ # A propriedade pode ser qualquer tipo de dado, inclusive OUTRO OBJETO.
+ # O construtor recebe o objeto pronto (injeção) e guarda numa propriedade.
+ class Turma:
+ def __init__(self, nome, professor):
+ self.nome = nome
+ self.professor = professor # outro objeto, criado fora
+ self.alunos = []
+ def matricular(self, aluno):
+ self.alunos.append(aluno)
+ class Professor:
+ def __init__(self, nome):
+ self.nome = nome
+ luan = Professor("Luan")
+ turma = Turma("ADS 2º semestre", luan)
+ turma.matricular(ana)
+ print(turma.professor.nome) # Luan
+ print(turma.alunos[0].nome) # Ana

Uma linha em vez de cinco

Sem construtor eram Aluno(), ana.nome = ..., ana.ra = ... e ana.notas = []. Com __init__, o objeto já nasce com todas as propriedades e nenhuma fica faltando.

Objeto dentro de objeto

A propriedade pode ser qualquer tipo, até outro objeto. Turma recebe o Professor pronto e guarda uma referência: ela não copia o professor.

Por que injetar

Receber o objeto pronto, em vez de criá-lo dentro da classe, deixa a Turma funcionar com qualquer Professor. Essa mesma ideia, passar o objeto pai ao construtor, é o que a árvore usa na próxima etapa.

saída
$ python3 construtor.py
Ana 8.5
Luan
Ana

arvore_arquivos.py: a árvore de arquivos

Uma única classe, Item, que vira pasta ou arquivo, e recebe o pai no construtor.

src/08/arvore_arquivos.py (arquivo novo)
+ # Árvore de arquivos: cada item conhece o seu pai.
+ # Rodar no terminal: python3 arvore_arquivos.py
+ #
+ # Uma pasta contém arquivos e outras pastas; cada item (menos a raiz) tem
+ # exatamente uma pasta pai. Esse encadeamento forma uma árvore, a mesma
+ # estrutura que o terminal mostra com o comando tree (Aula 01).
+ class Item:
+ # pai é um OBJETO Item (a pasta onde este item está) ou None (a raiz).
+ def __init__(self, nome, eh_pasta, pai=None):
+ # Só pasta pode conter itens: recusa já na criação.
+ if pai is not None and not pai.eh_pasta:
+ raise ValueError(pai.nome + " é um arquivo, não pode conter itens")
+ self.nome = nome
+ self.eh_pasta = eh_pasta
+ self.pai = pai
+ self.filhos = [] # um item só tem filhos se for pasta
+ # O item novo se cadastra na lista de filhos do pai.
+ if pai is not None:
+ pai.filhos.append(self)
+ # Sobe de pai em pai até a raiz, montando o caminho: igual ao pwd.
+ def caminho(self):
+ if self.pai is None:
+ return "/" + self.nome
+ return self.pai.caminho() + "/" + self.nome
+ # Quantos pais existem acima deste item. A raiz tem profundidade 0.
+ def profundidade(self):
+ if self.pai is None:
+ return 0
+ return self.pai.profundidade() + 1
+ # Mostra o item e, recursivamente, todos os filhos, recuando 4 espaços
+ # por nível de profundidade.
+ def mostrar(self):
+ recuo = " " * self.profundidade()
+ sufixo = ""
+ if self.eh_pasta:
+ sufixo = "/"
+ print(recuo + self.nome + sufixo)
+ for filho in self.filhos:
+ filho.mostrar()
+ # --- Montando a árvore -------------------------------------------------------
+ # Primeiro o pai, depois os filhos: o pai precisa existir para ser injetado.
+ raiz = Item("var", True)
+ www = Item("www", True, raiz)
+ unifaat = Item("unifaat", True, www)
+ src = Item("src", True, unifaat)
+ leia_me = Item("README.md", False, unifaat)
+ funcoes = Item("funcoes.py", False, src)
+ classes = Item("classes.py", False, src)
+ raiz.mostrar()
+ # var/
+ # www/
+ # unifaat/
+ # src/
+ # funcoes.py
+ # classes.py
+ # README.md
+ print(funcoes.caminho()) # /var/www/unifaat/src/funcoes.py
+ print(funcoes.profundidade()) # 4
+ print(len(src.filhos)) # 2
+ # Arquivo não aceita filhos: o construtor recusa com ValueError.
+ try:
+ Item("intruso", False, funcoes)
+ except ValueError as erro:
+ print(erro) # funcoes.py é um arquivo, não pode conter itens

Pai injetado, filhos cadastrados

__init__ recebe o objeto pai. Se ele existe, o item novo se cadastra em pai.filhos, usando o self para entregar a si mesmo. O pai precisa existir antes do filho.

Recursão subindo pelos pais

caminho() e profundidade() pedem a resposta ao pai e somam. A raiz (pai is None) é o caso base. É a recursão da Aula 07 e o pwd da Aula 01, feitos com objetos.

Sem else

Cada método trata a raiz com return antecipado e deixa o caso geral para o final. O if do construtor também só age quando há pai.
